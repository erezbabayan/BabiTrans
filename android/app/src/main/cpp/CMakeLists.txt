cmake_minimum_required(VERSION 3.22)
project(babiwhisper C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
# whisper.cpp pinned; set -DWHISPER_SRC=/path to use a local checkout
if(WHISPER_SRC AND EXISTS ${WHISPER_SRC}/CMakeLists.txt)
  FetchContent_Declare(whisper SOURCE_DIR ${WHISPER_SRC})
else()
  FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG 231638a0717b7822784297e5980965ffcec05810)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod+fp16" CACHE STRING "" FORCE)
FetchContent_MakeAvailable(whisper)
add_library(babiwhisper SHARED jni.cpp)
target_link_libraries(babiwhisper whisper log android)
